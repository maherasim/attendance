// ── This script creates its own Google Sheet automatically ───────
// ── No need to attach it to any sheet manually ──────────────────

var SHEET_NAME = "Attendance";

// ── getSheet: finds or creates the spreadsheet ───────────────────
// Uses PropertiesService to remember the sheet ID
// Uses SpreadsheetApp.create() — no DriveApp permission needed
function getSheet() {
  var props = PropertiesService.getScriptProperties();
  var ssId  = props.getProperty("SS_ID");
  var ss    = null;

  // Try to open existing spreadsheet
  if (ssId) {
    try { ss = SpreadsheetApp.openById(ssId); }
    catch(e) { ss = null; }
  }

  // If not found, create a brand new one
  if (!ss) {
    ss   = SpreadsheetApp.create("Vehari_UC_Attendance_Data");
    ssId = ss.getId();
    props.setProperty("SS_ID", ssId);

    // Rename default sheet and add headers
    var sheet = ss.getSheets()[0].setName(SHEET_NAME);
    var headers = [
      "ID", "Timestamp", "UC No", "UC Name", "Tehsil",
      "Secretary Name", "Latitude", "Longitude", "Accuracy (m)",
      "Distance (m)", "Google Maps", "Photo Link", "Server Time"
    ];
    sheet.appendRow(headers);
    sheet.getRange(1, 1, 1, headers.length)
         .setFontWeight("bold")
         .setBackground("#0d5c3a")
         .setFontColor("#ffffff");
    sheet.setFrozenRows(1);
    return sheet;
  }

  // Get or create the Attendance tab
  var sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    var headers = [
      "ID", "Timestamp", "UC No", "UC Name", "Tehsil",
      "Secretary Name", "Latitude", "Longitude", "Accuracy (m)",
      "Distance (m)", "Google Maps", "Photo Link", "Server Time"
    ];
    sheet.appendRow(headers);
    sheet.getRange(1, 1, 1, headers.length)
         .setFontWeight("bold")
         .setBackground("#0d5c3a")
         .setFontColor("#ffffff");
    sheet.setFrozenRows(1);
  }
  return sheet;
}

// ── doGet: returns all records as JSON or JSONP ──────────────────
function doGet(e) {
  try {
    var sheet   = getSheet();
    var data    = sheet.getDataRange().getValues();
    var records = [];

    if (data.length > 1) {
      var headers = data[0];
      for (var i = 1; i < data.length; i++) {
        var row = data[i];
        var obj = {};
        for (var j = 0; j < headers.length; j++) {
          obj[headers[j]] = row[j] !== undefined ? String(row[j]) : "";
        }
        records.push(obj);
      }
    }

    var result   = JSON.stringify({ success: true, records: records });
    var callback = e.parameter && e.parameter.callback;

    if (callback) {
      // JSONP response — wraps JSON in function call
      return ContentService
        .createTextOutput(callback + "(" + result + ")")
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }

    return ContentService
      .createTextOutput(result)
      .setMimeType(ContentService.MimeType.JSON);

  } catch(err) {
    var errResult = JSON.stringify({ success: false, error: err.toString(), records: [] });
    var callback  = e.parameter && e.parameter.callback;
    if (callback) {
      return ContentService
        .createTextOutput(callback + "(" + errResult + ")")
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return ContentService
      .createTextOutput(errResult)
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// ── doPost: saves one attendance record ─────────────────────────
function doPost(e) {
  try {
    var raw  = e.postData ? e.postData.contents : "{}";
    var data = JSON.parse(raw);
    var sheet = getSheet();

    sheet.appendRow([
      String(data.id             || ""),
      String(data.timestamp      || new Date().toISOString()),
      String(data.ucNumber       || ""),
      String(data.ucName         || ""),
      String(data.tehsil         || ""),
      String(data.secretaryName  || ""),
      data.lat                   || "",
      data.lng                   || "",
      data.locationAccuracy      || "",
      data.distanceFromOffice    || "",
      "https://maps.google.com/?q=" + (data.lat||"") + "," + (data.lng||""),
      "",
      new Date()
    ]);

    return ContentService
      .createTextOutput(JSON.stringify({ success: true }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({ success: false, error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// ── testSetup: run this ONCE from the editor to verify ───────────

}
